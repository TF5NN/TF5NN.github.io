{{- /* Turns whatever was pasted into the "YouTube video" box into the 11-character video ID. Accepts a bare ID or any usual YouTube address: dQw4w9WgXcQ https://www.youtube.com/watch?v=dQw4w9WgXcQ https://youtu.be/dQw4w9WgXcQ?si=abc https://www.youtube.com/shorts/dQw4w9WgXcQ https://www.youtube.com/embed/dQw4w9WgXcQ https://www.youtube.com/live/dQw4w9WgXcQ Returns "" if it cannot find an ID, and then no video is shown. */ -}} {{- $v := strings.TrimSpace (string .) }} {{- $id := "" }} {{- with findRESubmatch `(?:youtu\.be/|[?&]v=|/embed/|/shorts/|/live/)([A-Za-z0-9_-]{11})` $v 1 }} {{- $id = index (index . 0) 1 }} {{- else }} {{- if findRE `^[A-Za-z0-9_-]{11}$` $v }}{{ $id = $v }}{{ end }} {{- end }} {{- return $id }}
